// एडमिन पैनल डेटा लोड और सेफ मैनेजमेंट function loadAdminData() { let members = JSON.parse(localStorage.getItem('iois_all_members') || "[]"); const total = members.length; const approved = members.filter(m => m.status === "Activated").length; const pending = total - approved; document.getElementById('stat-total-users').innerText = total; document.getElementById('stat-pending').innerText = pending; document.getElementById('stat-approved').innerText = approved; const tbody = document.getElementById('members-table-body'); tbody.innerHTML = ""; if (members.length === 0) { tbody.innerHTML = `अभी तक कोई नया मेंबर रजिस्टर नहीं हुआ है।`; return; } members.forEach((m, index) => { const statusBadge = m.status === "Activated" ? `Activated` : `Pending`; const tr = document.createElement('tr'); tr.className = "hover:bg-gray-850 transition"; tr.innerHTML = ` ${m.name || 'N/A'} ${m.userid} ${m.phone || 'N/A'}
${m.email || 'N/A'} ${m.upi || 'N/A'} ${statusBadge} `; tbody.appendChild(tr); }); } // यूजर द्वारा अपलोड किया गया स्क्रीनशॉट देखने का सुरक्षित फंक्शन function viewUserScreenshot(userid) { let members = JSON.parse(localStorage.getItem('iois_all_members') || "[]"); const m = members.find(item => item.userid === userid); if (!m) { alert("डेटा नहीं मिला!"); return; } document.getElementById('ss-user-info').innerText = `Name: ${m.name} | User ID: ${m.userid}`; const imgPreview = document.getElementById('ss-img-preview'); // यदि यूजर ने इमेज अपलोड की है तो वह दिखेगी, अन्यथा डिफ़ॉल्ट मैसेज imgPreview.src = m.screenshot || "https://via.placeholder.com/400x300?text=No+Screenshot+Uploaded"; document.getElementById('screenshot-modal').classList.remove('hidden'); } function closeScreenshotModal() { document.getElementById('screenshot-modal').classList.add('hidden'); } // मेंबर अप्रूव करने का सुरक्षित तरीका (डेटा डिलीट नहीं होगा) function approveMember(userid) { let members = JSON.parse(localStorage.getItem('iois_all_members') || "[]"); let memberIndex = members.findIndex(item => item.userid === userid); if (memberIndex !== -1) { members[memberIndex].status = "Activated"; localStorage.setItem('iois_all_members', JSON.stringify(members)); // व्यक्तिगत यूजर ऑब्जेक्ट को भी अपडेट करें let rawU = localStorage.getItem("iois_user_" + userid); if (rawU) { let u = JSON.parse(rawU); u.status = "Activated"; localStorage.setItem("iois_user_" + userid, JSON.stringify(u)); } alert(`✅ मेंबर [${userid}] सफलतापूर्वक एक्टिवेट हो गया है!`); loadAdminData(); } }
